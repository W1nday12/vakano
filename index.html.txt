<!-- Interaction script -->
<script>
  // Header shadow on scroll
  const hdr = document.getElementById("site-header");
  const onScroll = () => {
    if (window.scrollY > 20) hdr.classList.add("scrolled");
    else hdr.classList.remove("scrolled");
  };
  document.addEventListener("scroll", onScroll, { passive: true });
  onScroll();

  // Modal
  const modalRoot = document.getElementById("modal-root");
  const formView = document.getElementById("modal-form-view");
  const successView = document.getElementById("modal-success-view");
  const form = document.getElementById("booking-form");
  const subjectSel = document.getElementById("f-subject");

  function openModal(subject) {
    if (subject) {
      const opts = Array.from(subjectSel.options);
      const match = opts.find(o => o.value === subject || o.textContent === subject);
      if (match) subjectSel.value = match.value;
      else {
        const o = document.createElement("option");
        o.textContent = subject;
        subjectSel.prepend(o);
        subjectSel.value = o.value;
      }
    }
    formView.style.display = "";
    successView.style.display = "none";
    modalRoot.classList.add("open");
    modalRoot.setAttribute("aria-hidden", "false");
    document.body.style.overflow = "hidden";
  }
  function closeModal() {
    modalRoot.classList.remove("open");
    modalRoot.setAttribute("aria-hidden", "true");
    document.body.style.overflow = "";
    setTimeout(() => form.reset(), 200);
  }
  document.querySelectorAll("[data-book]").forEach(b => {
    b.addEventListener("click", () => openModal(b.dataset.subject));
  });
  document.querySelectorAll("[data-close]").forEach(b => b.addEventListener("click", closeModal));
  document.addEventListener("keydown", e => { if (e.key === "Escape") closeModal(); });

  form.addEventListener("submit", e => {
    e.preventDefault();
    formView.style.display = "none";
    successView.style.display = "block";
    setTimeout(() => closeModal(), 3200);
  });

  // Reveal-on-scroll
  const io = new IntersectionObserver(entries => {
    entries.forEach(entry => {
      if (entry.isIntersecting) entry.target.classList.add("in");
    });
  }, { threshold: 0.12 });
  document.querySelectorAll("section.band > .wrap > *, .hero-inner > *, .tour, .about-card").forEach((el, i) => {
    el.classList.add("reveal");
    el.style.transitionDelay = (i % 6) * 40 + "ms";
    io.observe(el);
  });
</script>
